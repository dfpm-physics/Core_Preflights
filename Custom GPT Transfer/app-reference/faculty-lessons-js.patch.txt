+/**
+ * Read-only catalog used by the generated-link destination picker.
+ * Includes only the metadata needed to classify an attachment and capture
+ * compare-and-swap baselines; existing private Markdown is never loaded here.
+ */
+export async function loadDestinationLessons(courseId, incomingInteractionId) {
+  const { data: lessonRows, error: lessonError } = await db.from('lessons')
+    .select(LESSON_COLS).eq('course_id', courseId)
+    .order('lesson_number', { nullsFirst: false }).order('title');
+  if (lessonError) return { lessons: [], error: lessonError };
+
+  const lessons = lessonRows || [];
+  const lessonIds = lessons.map(l => l.id);
+  const preflightIds = [...new Set(lessons.map(l => l.preflight_id).filter(Boolean))];
+  const interactionIds = [...new Set(lessons.map(l => l.interaction_id).filter(Boolean))];
+  const reportChecks = interactionIds.map(async interactionId => {
+    const { count, error } = await db.from('preflight_interaction_reports')
+      .select('interaction_id', { count: 'exact', head: true }).eq('interaction_id', interactionId);
+    return { interactionId, count: count || 0, error };
+  });
+
+  const [assignmentResult, interactionResult, inputResult, reportResult, ownerResult, incomingResult] =
+    await Promise.all([
+      preflightIds.length
+        ? db.from('assignments').select('id, title, description, questions, is_published').in('id', preflightIds)
+        : Promise.resolve({ data: [], error: null }),
+      interactionIds.length
+        ? db.from('interactions').select('id, course_id, title, description, artifact_url, is_published').in('id', interactionIds)
+        : Promise.resolve({ data: [], error: null }),
+      lessonIds.length
+        ? db.from('lesson_chat_inputs').select('lesson_id, version, source_filename, updated_at').in('lesson_id', lessonIds)
+        : Promise.resolve({ data: [], error: null }),
+      Promise.all(reportChecks),
+      db.from('lessons').select('id, course_id, title, interaction_id')
+        .eq('interaction_id', incomingInteractionId).maybeSingle(),
+      db.from('interactions').select('id, course_id').eq('id', incomingInteractionId).maybeSingle(),
+    ]);
+
+  const firstError = [
+    assignmentResult.error, interactionResult.error, inputResult.error,
+    ...reportResult.map(r => r.error),
+    ownerResult.error, incomingResult.error,
+  ].find(Boolean);
+  if (firstError) return { lessons: [], error: firstError };
+
+  const assignmentsById = Object.fromEntries((assignmentResult.data || []).map(a => [a.id, a]));
+  const interactionsById = Object.fromEntries((interactionResult.data || []).map(i => [i.id, i]));
+  const inputsByLesson = Object.fromEntries((inputResult.data || []).map(i => [i.lesson_id, i]));
+  const reportCounts = {};
+  reportResult.forEach(r => { reportCounts[r.interactionId] = r.count; });
+
+  return {
+    error: null,
+    incomingOwner: ownerResult.data || null,
+    incomingInteraction: incomingResult.data || null,
+    lessons: lessons.map(l => ({
+      ...l,
+      objectives: Array.isArray(l.objectives) ? l.objectives : [],
+      preflight: l.preflight_id ? (assignmentsById[l.preflight_id] || null) : null,
+      interaction: l.interaction_id ? (interactionsById[l.interaction_id] || null) : null,
+      chat_input: inputsByLesson[l.id] || null,
+      interaction_report_count: l.interaction_id ? (reportCounts[l.interaction_id] || 0) : 0,
+    })),
+  };
+}
